function useCareMutation<TVars, TRes = unknown>(hid: string | undefined, fn: (v: TVars) => Promise<TRes>) {
  const inv = useInvalidateCare();
  return useMutation({ mutationFn: fn, onSuccess: () => inv(hid) });
}
