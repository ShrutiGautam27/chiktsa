export function EngineBadge({ engine }: { engine: Engine }) {
  return engine === "gemini"
    ? <Badge tone="info" title="Generated by Google Gemini from the cited records"><Cloud className="size-3" /> Gemini</Badge>
    : <Badge tone="neutral" title="Generated by Chiktsa's built-in rules — no cloud AI used"><Cpu className="size-3" /> Offline rules</Badge>;
}

export function BriefingCard({ hid }: { hid: string }) {
  const b = useBriefing(hid);
  const refresh = useRefreshBriefing(hid);
  const data = refresh.data ?? b.data;
  return (
    <Card className="overflow-hidden">
      <div className="bg-gradient-to-br from-brand-700 to-brand-900 p-5 text-white">
        <div className="flex items-center justify-between gap-2">
          <span className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wider text-brand-100"><Sparkles className="size-3.5" /> Today's care briefing</span>
          <button onClick={() => refresh.mutate(undefined, { onError: (e) => toast.error(errorMessage(e)) })} disabled={refresh.isPending}
            className="rounded-lg p-1.5 text-brand-100 hover:bg-white/10" aria-label="Refresh briefing"><RefreshCw className={cn("size-4", refresh.isPending && "animate-spin")} /></button>
        </div>
        {b.isLoading ? <Skeleton className="mt-3 h-12 bg-white/10" /> : data && <>
          <p className="mt-2 font-display text-xl">{data.headline}</p>
          <p className="mt-1 text-sm text-white/85">{data.summary}</p>
        </>}
      </div>
      <CardContent className="pt-4">
        {b.isError ? <ErrorState error={b.error} onRetry={() => b.refetch()} /> : data && <>
          {data.priorities.length > 0 && <ul className="space-y-2">{data.priorities.map((p) => (

// ------------------------------------------------------------------ Task converter
export function TaskConverterDialog({ hid, open, onOpenChange }: { hid: string; open: boolean; onOpenChange: (o: boolean) => void }) {
  const convert = useConvertTasks(hid);
  const create = useCreateTask(hid);
  const [text, setText] = useState("");
  const [items, setItems] = useState<(TaskProposal & { keep: boolean; dueLocal: string })[]>([]);
  const toLocal = (iso: string | null) => (iso ? new Date(new Date(iso).getTime() - new Date().getTimezoneOffset() * 60000).toISOString().slice(0, 16) : "");
  const run = () => convert.mutate(text, {
    onSuccess: (r) => setItems(r.proposals.map((p) => ({ ...p, keep: true, dueLocal: toLocal(p.dueAt) }))),
    onError: (e) => toast.error(errorMessage(e)),
  });
  const save = async () => {
    const chosen = items.filter((i) => i.keep && i.title.trim());
    try {
      for (const i of chosen) {
        await create.mutateAsync({ title: i.title.trim(), seniorId: i.seniorId, kind: i.kind, assigneeUid: i.assigneeUid,
          dueAt: i.dueLocal ? new Date(i.dueLocal).toISOString() : null, priority: "normal", description: "" });
      }
      toast.success(`${chosen.length} task(s) created`);
      onOpenChange(false); setText(""); setItems([]);
    } catch (e) {
      toast.error(errorMessage(e));
    }
  };
  const upd = (i: number, patch: Partial<(typeof items)[number]>) => setItems(items.map((x, j) => (j === i ? { ...x, ...patch } : x)));
  return (
    <Dialog open={open} onOpenChange={onOpenChange} className="max-w-2xl" title="Describe it in your own words"
      description="English, Hindi or Hinglish — e.g. “Kal Priya Dadi ko eye doctor le jayegi 5 baje, and buy groceries today”. You confirm every task before it's saved.">
      <div className="space-y-4">
        <Textarea rows={3} value={text} onChange={(e) => setText(e.target.value)} aria-label="Describe the tasks" placeholder="What needs doing?" />
        <div className="flex justify-end"><Button onClick={run} loading={convert.isPending} disabled={text.trim().length < 3}><Wand2 /> Suggest tasks</Button></div>
        {convert.data && (
          <div className="space-y-3">
            <div className="flex items-center gap-2 text-xs text-ink-500"><EngineBadge engine={convert.data.engine} /> {convert.data.note}</div>
            {!items.length && <p className="text-sm text-ink-500">No tasks found — try describing who should do what, and when.</p>}
            {items.map((it, i) => (
              <div key={i} className={cn("space-y-3 rounded-xl border p-3", it.keep ? "border-line" : "border-dashed opacity-60")}>
                <label className="flex items-center gap-2 text-sm"><input type="checkbox" className="size-4 accent-brand-600" checked={it.keep} onChange={(e) => upd(i, { keep: e.target.checked })} /> Create this task</label>
                <Field label="Title"><Input value={it.title} onChange={(e) => upd(i, { title: e.target.value })} /></Field>
                <div className="grid gap-3 sm:grid-cols-2">
                  <Field label="Assign to"><MemberSelect hid={hid} seniorId={it.seniorId} value={it.assigneeUid ?? ""} onChange={(v) => upd(i, { assigneeUid: v || null })} placeholder="Leave unassigned" /></Field>
                  <Field label="Due"><Input type="datetime-local" value={it.dueLocal} onChange={(e) => upd(i, { dueLocal: e.target.value })} /></Field>
                </div>
                <p className="text-xs text-ink-500">For: {it.seniorName ?? "the household"} · type: {it.kind.replace("_", " ")}</p>
                {it.needsClarification && it.clarification && <p className="rounded-lg bg-warning-50 px-3 py-2 text-xs text-warning-600">{it.clarification}</p>}
              </div>
            ))}
            {items.length > 0 && <div className="flex justify-end"><Button onClick={save} loading={create.isPending}>Create selected tasks</Button></div>}
          </div>
        )}
      </div>
    </Dialog>
  );
}
 
// ------------------------------------------------------------------ Care notes
const TAGS = ["mood", "meals", "sleep", "mobility", "activity", "social"];
 
export function CareNotesCard({ hid, sid, name }: { hid: string; sid: string; name: string }) {
  const notes = useNotes(hid, sid);
  const add = useAddNote(hid, sid);
  const { can } = useCareContext();
  const [text, setText] = useState("");
  const [tags, setTags] = useState<string[]>([]);
  if (notes.isError && notes.error instanceof ApiError && notes.error.status === 403)
    return <Card className="p-5 text-sm text-ink-500">{name} hasn't shared care notes with you.</Card>;
  return (
    <Card>
      <CardHeader><CardTitle className="flex items-center gap-2"><NotebookPen className="size-4 text-brand-600" /> Daily observations</CardTitle>
        <CardDescription>Everyday, non-clinical notes — they also help the Care Memory answer questions later.</CardDescription></CardHeader>
      <CardContent className="space-y-4">
        {can("care.log.write") && (
          <form className="space-y-2" onSubmit={(e) => { e.preventDefault(); if (!text.trim()) return;
            add.mutate({ text, tags }, { onSuccess: () => { setText(""); setTags([]); toast.success("Observation saved"); }, onError: (er) => {
              if (!isOfflineError(er)) return toast.error(errorMessage(er));
              enqueue({ type: "note.create", householdId: hid, seniorId: sid, label: "Observation", payload: { text, tags } });
              setText(""); setTags([]); toast.info("You're offline — observation saved on this device and will sync.");
            } }); }}>
            <Textarea rows={2} value={text} onChange={(e) => setText(e.target.value)} placeholder={`How is ${name} today?`} aria-label="New observation" maxLength={2000} />
            <div className="flex flex-wrap items-center gap-1.5">
              {TAGS.map((t) => <button key={t} type="button" aria-pressed={tags.includes(t)} onClick={() => setTags(tags.includes(t) ? tags.filter((x) => x !== t) : [...tags, t])}
                className={cn("rounded-full border px-2.5 py-0.5 text-xs capitalize", tags.includes(t) ? "border-brand-500 bg-brand-50 text-brand-800" : "border-line text-ink-500")}>{t}</button>)}
              <Button type="submit" size="sm" className="ml-auto" loading={add.isPending} disabled={!text.trim()}>Save</Button>
            </div>
          </form>
        )}
        {notes.isLoading ? <Skeleton className="h-20" /> : notes.data?.length ? (
          <ul className="space-y-2">{notes.data.slice(0, 8).map((n) => (
            <li key={n.id} className="rounded-xl bg-sage-50 p-3 text-sm"><p>{n.text}</p>
              <p className="mt-1 text-xs text-ink-500">{n.authorName} · {relativeTime(n.createdAt)}{n.tags.length > 0 && ` · ${n.tags.join(", ")}`}</p></li>))}</ul>
        ) : <p className="text-sm text-ink-500">No observations yet.</p>}
      </CardContent>
    </Card>
  );
}
 
// ------------------------------------------------------------------ AI settings (opt-in)
export function AISettingsCard({ hid }: { hid: string }) {
  const st = useAIStatus(hid);
  const set = useSetAIMode(hid);
  const { can } = useCareContext();
  if (st.isLoading) return <Skeleton className="h-40" />;
  if (st.isError) return <ErrorState error={st.error} />;
  const s = st.data!;
  return (
    <Card>
      <CardHeader><CardTitle className="flex items-center gap-2"><ShieldCheck className="size-4 text-brand-600" /> AI & privacy</CardTitle>
        <CardDescription>Choose whether care records may be sent to Google Gemini for AI features. Offline mode never sends data outside Chiktsa.</CardDescription></CardHeader>
      <CardContent className="space-y-4 text-sm">
        <div role="radiogroup" aria-label="AI processing" className="grid gap-2 sm:grid-cols-2">
          {([["offline", "Offline rules", "Built-in, deterministic. No data leaves Chiktsa.", Cpu], ["gemini", "Cloud AI (Gemini)", "Richer answers and photo-prescription reading. Only the minimal records for one senior are sent per request.", Cloud]] as const).map(([v, l, d, I]) => (
            <button key={v} type="button" role="radio" aria-checked={s.mode === v} disabled={!can("household.manage") || set.isPending}
              onClick={() => set.mutate(v, { onSuccess: () => toast.success(v === "gemini" ? "Cloud AI enabled" : "Switched to offline mode"), onError: (e) => toast.error(errorMessage(e)) })}
              className={cn("rounded-xl border p-3 text-left disabled:cursor-not-allowed", s.mode === v ? "border-brand-500 bg-brand-50" : "border-line hover:bg-sage-50")}>
              <span className="flex items-center gap-2 font-medium"><I className="size-4" /> {l}</span><span className="mt-1 block text-xs text-ink-500">{d}</span>
            </button>
          ))}
        </div>
        {s.mode === "gemini" && !s.geminiConfigured && <p className="rounded-xl bg-warning-50 p-3 text-warning-600">Cloud AI is enabled for this household, but no Gemini API key is configured on the server yet — features keep running in offline mode until it is.</p>}
        <p className="text-xs text-ink-500">Currently using: <strong>{s.effectiveEngine === "gemini" ? `Gemini (${s.model})` : "offline rules"}</strong>. AI never diagnoses, never changes medication and every AI action is logged without its content.
          {!can("household.manage") && " Only household administrators can change this."}</p>
      </CardContent>
    </Card>
  );
}
